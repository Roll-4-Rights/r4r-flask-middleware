CREATE TABLE users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    first_name VARCHAR(100) NOT NULL,
    last_name VARCHAR(100) NOT NULL,
    user_type VARCHAR(20) DEFAULT 'donator' CHECK (user_type IN ('donator', 'admin')),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_users_email ON users(email);
CREATE INDEX idx_users_user_type ON users(user_type);

# Flask Backend - Holds the secret token
@app.route('/api/items', methods=['GET'])
@login_required
def get_items():
    user_id = session['user_id']
    
    # Flask makes request to NocoDB with SECRET token
    response = requests.get(
        f'{NOCODB_URL}/api/v1/db/data/noco/{BASE_ID}/donation_items',
        headers={'xc-auth': NOCODB_SECRET_TOKEN},  # Hidden from frontend!
        params={'where': f'(user_id,eq,{user_id})'}
    )
    
    return jsonify(response.json())
